
function nav(active){
  let out = "";
  GROUPS_NAV.forEach(function(g, gi){
    var label = g[0], items = g[1];
    out += '\n        <div style="font-size:10.5px;font-weight:700;letter-spacing:.12em;color:var(--ink3);padding:' + (gi===0?"0 8px 6px":"16px 8px 6px") + '">' + label + '</div>';
    items.forEach(function(it){
      var name = it[0], href = it[1], icon = it[2], badge = it[3];
      var on = name === active;
      var st = on
        ? 'display:flex;align-items:center;gap:9px;padding:9px 12px;border-radius:10px;color:var(--grnink);font-weight:700;background:var(--grnsoft);box-shadow:0 1px 2px rgba(34,34,31,.05)'
        : 'display:flex;align-items:center;gap:9px;padding:9px 12px;border-radius:10px';
      var b = badge ? '<span style="font-weight:700;font-size:12.5px;color:var(--' + (badge[1]==="grn"?"grnink":"amb") + ')">' + badge[0] + '</span>' : "";
      out += '\n        <a href="' + (on?"#":href) + '" style="' + st + '"><svg width="16" height="16" viewBox="0 0 16 16" fill="none" stroke="currentColor" stroke-width="1.4">' + icon + '</svg><span style="flex:1">' + name + '</span>' + b + '</a>';
    });
  });
  return out;
}
